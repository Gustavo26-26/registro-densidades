// ¡PEGA AQUÍ TU URL DE APPS SCRIPT! const urlGoogleScript = "https://script.google.com/macros/s/AKfycbyvsDKF4T-Ge4wLAyEoyxS9El2lHlP-El62zbMUBXNc_QKJjbby3y9Lm4VnOsaXrlZ6/exec"; let datosGlobales = []; let graficoInstance = null; async function cargarDatos() { try { document.getElementById('tablaDetalle').innerHTML = 'Cargando datos desde la mina...'; const respuesta = await fetch(urlGoogleScript); datosGlobales = await respuesta.json(); llenarFiltros(); aplicarFiltros(); } catch (error) { console.error("Error cargando datos:", error); alert("Error al conectar con la base de datos."); } } function llenarFiltros() { // Extraemos valores únicos filtrando celdas vacías const niveles = [...new Set(datosGlobales.map(d => d['Nivel']))].filter(Boolean); const proyectos = [...new Set(datosGlobales.map(d => d['Proyecto']))].filter(Boolean); const selectNivel = document.getElementById('filtroNivel'); const selectProyecto = document.getElementById('filtroProyecto'); // Llenamos los selects selectNivel.innerHTML = '' + niveles.map(n => ``).join(''); selectProyecto.innerHTML = '' + proyectos.map(p => ``).join(''); } function aplicarFiltros() { const nivelSel = document.getElementById('filtroNivel').value; const proyectoSel = document.getElementById('filtroProyecto').value; let filtrados = datosGlobales; // FILTRO ULTRA-ROBUSTO PARA NIVEL if (nivelSel !== "TODOS") { filtrados = filtrados.filter(d => { if (d['Nivel'] == null) return false; const valorHoja = String(d['Nivel']).trim().toLowerCase(); const valorFiltro = String(nivelSel).trim().toLowerCase(); return valorHoja === valorFiltro; }); } // FILTRO ULTRA-ROBUSTO PARA PROYECTO if (proyectoSel !== "TODOS") { filtrados = filtrados.filter(d => { if (d['Proyecto'] == null) return false; const valorHoja = String(d['Proyecto']).trim().toLowerCase(); const valorFiltro = String(proyectoSel).trim().toLowerCase(); return valorHoja === valorFiltro; }); } procesarDashboard(filtrados); } function procesarDashboard(datos) { let aprobados = 0; let desviados = 0; let filasTabla = ''; const labelsTaladros = []; const dataDensidad = []; datos.forEach(d => { const densidad = parseFloat(d['Densidad Medida']); const rangoStr = d['Rango Teórico']; if(isNaN(densidad) || !rangoStr) return; const partes = String(rangoStr).split('-'); const min = parseFloat(partes[0].trim()); const max = parseFloat(partes[1].trim()); let estadoHtml = ''; if (densidad >= min && densidad <= max) { aprobados++; estadoHtml = '🟢 OK'; } else { desviados++; estadoHtml = '🔴 DESVIADO'; } labelsTaladros.push(`Tal-${d['Taladro']}`); dataDensidad.push(densidad); filasTabla += ` ${d['Taladro']} ${densidad.toFixed(3)} ${estadoHtml} `; }); document.getElementById('kpiTotal').innerText = aprobados + desviados; document.getElementById('kpiAprobados').innerText = aprobados; document.getElementById('kpiDesviados').innerText = desviados; document.getElementById('tablaDetalle').innerHTML = filasTabla || 'No hay datos para este filtro'; dibujarGrafico(labelsTaladros, dataDensidad); } function dibujarGrafico(labels, data) { const ctx = document.getElementById('graficoTendencia').getContext('2d'); if (graficoInstance) { graficoInstance.destroy(); } graficoInstance = new Chart(ctx, { type: 'line', data: { labels: labels, datasets: [{ label: 'Densidad Medida', data: data, borderColor: 'rgb(59, 130, 246)', backgroundColor: 'rgba(59, 130, 246, 0.1)', borderWidth: 2, pointRadius: 4, pointBackgroundColor: 'rgb(30, 64, 175)', fill: true, tension: 0.3 }] }, options: { responsive: true, scales: { y: { title: { display: true, text: 'Densidad (g/cc)' } }, x: { title: { display: true, text: 'N° Taladro' } } } } }); } window.onload = cargarDatos;
